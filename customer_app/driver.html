// عرض الطلبات في لوحة الكابتن مع بيانات وموقع العميل جاهزة function displayCaptainOrders(ordersList) { const container = document.getElementById('captain-orders-container'); if (!container) return; container.innerHTML = ''; ordersList.forEach(order => { const mapsUrl = order.location && order.location.lat ? `https://www.google.com/maps?q=${order.location.lat},${order.location.lng}` : `https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(order.customerAddress || '')}`; const orderCard = document.createElement('div'); orderCard.className = 'order-card-captain'; orderCard.innerHTML = `

طلب جديد من: ${order.customerName || 'عميل'}

الهاتف: ${order.customerPhone || 'غير متوفر'}

العنوان: ${order.customerAddress || 'تحديد مباشر عبر الخريطة'}

📍 فتح الموقع على الخريطة للكابتن `; container.appendChild(orderCard); }); }
